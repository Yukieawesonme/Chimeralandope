Arquitetura TypeScript

ECS, Object Pooling, Fixed Timestep, EventBus desacoplado

// Core ECS - Entity Component System
class Entity {
    readonly id: number;
    private components: Map = new Map();

    addComponent(comp: T): T { ... }
    getComponent(ctor: new ...args: any[] => T): T | null { ... }
}

// Fixed Timestep Game Loop (60 FPS)
class Engine {
    fixedDeltaTime = 1 / 60;
    accumulator = 0;

    gameLoop() {
        const dt = this.getDeltaTime();
        this.accumulator += dt;
        while (this.accumulator >= this.fixedDeltaTime) {
            this.fixedUpdate(this.fixedDeltaTime);
            this.accumulator -= this.fixedDeltaTime;
        }
        this.update(dt);
    }
}

// Object Pooling - Zero GC
class ObjectPool {
    private pool: T[] = [];
    private active: Set = new Set();

    get(): T { // Reuse or create }
    release(obj: T): void { // Reset & return }
}

// EventBus - Decoupled Communication
const EventBus = {
    on(event: string, cb: (data: T) => void),
    emit(event: string, data: T),
    once(event: string, cb: (data: T) => void)
};

// Weapon Switch Cancel System
class WeaponSystem {
    slots: WeaponSlotData[] = [...];
    activeSlot = 0;

    requestSwitch(slot: number) {
        const current = this.getActiveWeapon();
        if (current.canSwitchCancel()) {
            current.onAttackEnded(); // Cancel recovery
            const next = this.slots[slot].weapon;
            next.onSwitchCancel(current); // Switch attack!
        }
    }
}
